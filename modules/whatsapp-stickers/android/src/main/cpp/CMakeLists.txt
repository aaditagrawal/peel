cmake_minimum_required(VERSION 3.22.1)
project(peelwebp LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO WEBPMUX EXTRAS)
  set(WEBP_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(WEBP_BUILD_LIBWEBPMUX ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/libwebp-1.6.0)
add_library(peelwebp SHARED encoder.cpp)
target_include_directories(peelwebp PRIVATE vendor/libwebp-1.6.0/src)
target_link_libraries(peelwebp webp webpdemux libwebpmux jnigraphics)
target_link_options(peelwebp PRIVATE "-Wl,-z,max-page-size=16384")
